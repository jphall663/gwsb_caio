\section{Extraction \& Verification}

\begin{frame}{1. Raw Triplet Extraction}
\begin{itemize}
  \item \textbf{Goal:} Quickly preprocess raw text data and ensure the AI remains grounded in the actual text.
  \item \textbf{Prompt:} \texttt{Hi, please read the attached document. Then extract 50 entity-attribute-relationship triplets from the attached document, each representing a key sentence or phrase. Please present the triplets in the order the sentences or phrases occur in the text. Present the triplets with a sequence number but without headings or classifications.}
  \item \textbf{Hint:} Do not mention the name or specific topic of the document to help ensure the extraction is based on your document, not scraped training data.
\end{itemize}
\end{frame}

\begin{frame}{2. Spot-Check Accuracy}
\begin{itemize}
  \item \textbf{Goal:} Verify that the triplets are not hallucinations by linking them back to source sentences.
  \item \textbf{Prompts:}
    \begin{itemize}
      \item \texttt{Thanks. Now I would like to check that these triplets accurately reflect the document. Please state the sentence or phrase from the document that the first triplet represents.}
      \item \texttt{Thanks. Please state the sentence or phrase from the document that the 25th triplet represents.}
      \item \texttt{Thanks. Please state the sentence or phrase from the document that the last triplet represents.}
    \end{itemize}
\end{itemize}
\end{frame}

% =========================
% #### Thematic Organization (SECTION)

\section{Thematic Organization}

\begin{frame}{3. Thematic Clustering}
\begin{itemize}
  \item \textbf{Goal:} Move from a sequential list to a structured overview.
  \item \textbf{Prompt:} \texttt{Thanks. Now please perform a thematic clustering of the triplets. Present the cluster labels and the triplets that represent each cluster. Print each triplet with its original sequence number. Print the total number of triplets in each cluster.}
  \item \textbf{Check:}
    \begin{itemize}
      \item Ensure triplets link back to their original sequence number.
      \item Verify cluster subtotals, and confirm a total of 50 triplets.
    \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{4. Natural Language Conversion}
\begin{itemize}
  \item \textbf{Goal:} Transform abbreviated triplets into readable summary notes.
  \item \textbf{Prompt:} \texttt{Thanks. Now reprint the cluster labels and the triplets, but please convert the triplets into plain English bullets or short sentences. Preserve the total number of bullets and the number in each cluster from the thematic clustering. Print the original sequence number of the triplet at the end of each bullet.}
  \item \textbf{Check:}
    \begin{itemize}
      \item Ensure bullets link back to their original sequence number.
      \item Verify cluster subtotals, and confirm a total of 50 bullets.
    \end{itemize}
\end{itemize}
\end{frame}

\section{Exporting to Technical Formats}

\begin{frame}{5. LaTeX Beamer Slide Deck}
\begin{itemize}
  \item \textbf{Goal:} Use the structured data to generate a presentation asset.
  \item \textbf{Prompt:} \texttt{Thanks. Now, please generate a short LaTeX Beamer slide deck from the cluster labels and the bullets. The cluster labels should become section headers and slide titles. Please preserve the numbering of the bullets, and ensure each cluster is represented in the slide deck and that the deck contains the correct total number of bullets.}
\end{itemize}
\end{frame}

\begin{frame}{6. HTML Web Page}
\begin{itemize}
  \item \textbf{Goal:} Use the structured data to generate a basic web asset.
  \item \textbf{Prompt:} \texttt{Thanks. Now, please generate a basic HTML page from the cluster labels and the bullets. The cluster labels should become section headers. Please preserve the numbering of the bullets, and ensure each cluster is represented in the page and that the page contains the correct total number of bullets.}
  \item \textbf{Check:}
    \begin{itemize}
      \item Ensure bullets link back to their original sequence number.
      \item Verify cluster subtotals, and confirm a total of 50 bullets.
    \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{7. Microsoft Office Documents}
\begin{itemize}
  \item \textbf{Goal:} Use the structured data to generate a PowerPoint or Word document.
  \item \textbf{Prompt:} \texttt{Thanks. Now, can you generate a PowerPoint or Word file?} (If yes ...)
  \item \textbf{Prompt:} \texttt{Great! Now, please generate a Word document/PowerPoint slides from the cluster labels and the bullets. The cluster labels should become section headers. Please preserve the numbering of the bullets, and ensure each cluster is represented in the document and that the document contains the correct total number of bullets.} 
  \item \textbf{Check:}
    \begin{itemize}
      \item Ensure bullets link back to their original sequence number.
      \item Verify cluster subtotals, and confirm a total of 50 bullets.
    \end{itemize}
  \item \textbf{Hint:} Works best on Copilot. Or just copy and paste text from chatbot output into a Word or PowerPoint document.
\end{itemize}
\end{frame}

\section{Assessment}

\begin{frame}{8. Question Generation}
\begin{itemize}
  \item \textbf{Goal:} Create 50 short-answer questions from the entity and relationship aspects of the triplets.
  \item \textbf{Prompt:} \texttt{Thanks. Now please use the entity and relationship aspects of the triplets to create 50 short-answer questions. Please print the questions with their cluster labels. Preserve the total number of bullets and the number in each cluster from the thematic clustering. Print the original sequence number of the triplet at the end of each question.}
\end{itemize}
\end{frame}

\begin{frame}{9. Creating the Answer Key}
\begin{itemize}
  \item \textbf{Goal:} Print each question with its corresponding answer bullet using the previously created summary bullets to create question/answer pairs.
  \item \textbf{Prompt:} \texttt{Thanks. Now please print each question with its corresponding answer bullet using the previously created summary bullets to create question/answer pairs. Preserve the total number of questions and answers and the number of pairs in each cluster from the thematic clustering. Print the original sequence number of the triplet at the end of each answer.}
  \item \textbf{Check:}
    \begin{itemize}
      \item Ensure Q\&A pairs match one another and link back to their original sequence number.
      \item Verify cluster subtotals, and confirm a total of 50 Q\&A pairs.
    \end{itemize}
  \item \textbf{Hint:} You now also have a benchmark dataset for assessing the capability of an language model to answer questions about your document.
\end{itemize}
\end{frame}

\section{Interactive Visualization}

\begin{frame}{10. Python Word Cloud (Colab)}
\begin{itemize}
  \item \textbf{Goal:} Generate straightforward Colab code to print a word cloud of the original 50 triplets.
  \item \textbf{Prompt:} \texttt{Thanks. Now please show me the most straightforward Python code I can use in a Colab notebook to print a word cloud of the original 50 triplets. Please include all 50 triplets as a Python list of lists in the generated example code. Include the original sequence number as a comment after each individual list. Please include any necessary !pip install statements at the beginning of the code example.}
  \item \textbf{Check:}
    \begin{itemize}
      \item Confirm triplets in Python match original triplets.
      \item Test the Python code.
    \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{11. Mermaid Knowledge Graph}
\begin{itemize}
  \item \textbf{Goal:} Use the first cluster of triplets to create a knowledge graph in Mermaid. 
  \item \textbf{Prompt:} \texttt{Thanks. Now please use the first cluster of triplets to create a knowledge graph in Mermaid. Please print the text that defines the Mermaid graph, and then separately provide instructions for how to view and download it. Retain the cluster label as the graph title and sequence number of the triplets as comments in Mermaid.}
  \item \textbf{Check:} Confirm Mermaid triplets match the original first cluster triplets.
\end{itemize}
\end{frame}

\begin{frame}{12. Visual Integration (Image Handling)}
\begin{itemize}
  \item \textbf{Goal:} Use generated code snippets to integrate images into LaTeX and HTML documents. 
  \item \textbf{Prompts:}
    \begin{itemize}
      \item \texttt{What is the easiest way to display a static image from the Colab notebook or Mermaid in the LaTeX generated earlier?}
      \item \texttt{What is the easiest way to display a static image from the Colab notebook or Mermaid in the HTML generated earlier?}    
    \end{itemize}
  \item \textbf{Hint:} Copy and paste the visuals into Office documents.
\end{itemize}
\end{frame}
